def convolve_four_loops(image, kernel):
    image, flipped, padded = _prepare(image, kernel)
    kh, kw = flipped.shape
    output = np.zeros_like(image, dtype=np.float64)

    for row in range(image.shape[0]):
        for col in range(image.shape[1]):
            for krow in range(kh):
                for kcol in range(kw):
                    output[row, col] += (
                        padded[row + krow, col + kcol]
                        * flipped[krow, kcol]
                    )
    return output
def convolve_two_loops(image, kernel):
    image, flipped, padded = _prepare(image, kernel)
    kh, kw = flipped.shape
    output = np.zeros_like(image, dtype=np.float64)

    for row in range(image.shape[0]):
        for col in range(image.shape[1]):
            patch = padded[row:row + kh, col:col + kw]
            output[row, col] = np.sum(patch * flipped)
    return output
